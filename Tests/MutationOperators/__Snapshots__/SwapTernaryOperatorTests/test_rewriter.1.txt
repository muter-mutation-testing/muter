func someCode(_ a: Bool, _ b: Bool) -> Bool {
  if __SwiftMutator.activeMutant == 0 {
    return a ? false : b ? true : false
  } else if __SwiftMutator.activeMutant == 0 {
    return a ? b ? false : true : false
  } else {
    return a ? b ? true : false : false
  }
}
private enum __SwiftMutator {
  nonisolated static let activeMutant = ProcessInfo.processInfo.environment["MUTER_ACTIVE_MUTANT"]
    .flatMap { Int($0) }
}
