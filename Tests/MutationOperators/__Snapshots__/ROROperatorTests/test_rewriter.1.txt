internal func isBare() throws -> Bool {
  return try self.cachedIsBareRepo.memoize(body: {
    if __SwiftMutator.activeMutant == 0 {
      let output = try callGit(
        "rev-parse",
        "--is-bare-repository",
        failureMessage: "Couldn’t test for bare repository"
      )

      return output != "true"
    } else {
      let output = try callGit(
        "rev-parse",
        "--is-bare-repository",
        failureMessage: "Couldn’t test for bare repository"
      )

      return output == "true"
    }
  })
}
